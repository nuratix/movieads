Movieflix React Native / Web Integration Snippet 
// Fetch active 16:9 ads in Movieflix
export async function fetchMovieAds() {
  const res = await fetch('https://ads.wallpaperdomain.xyz/api/v1/ads', {
    headers: {
      'X-Ad-Secret-Key': process.env.ADS_SECRET_KEY || 'mf_ad_sec_89a31bc09e',
    },
  });
  if (!res.ok) return [];
  const data = await res.json();
  return data.ads || [];
}
